# Air offsite
hom_add_air_offsite_effect = {
	add_offsite_building = { type = industrial_complex level = 15 }
	add_offsite_building = { type = arms_factory level = 15 }
	add_offsite_building = { type = synthetic_refinery level = 10 }
	
	set_temp_variable = {
		max_iteration = 3
	}
	while_loop_effect = {
		limit = {
			check_variable = {
				var = max_iteration
				value = 0
				compare = greater_than
			}
		}
		random_owned_controlled_state = {
			limit = {
				is_core_of = ROOT
				hom_has_facility_building = no
			}
			construct_building_in_random_province = {
				air_facility = 1
			}
		}
		subtract_from_temp_variable = {
			max_iteration = 1
		}
	}
}

# For naval treaty removal if possible
hom_remove_naval_treaty_effect = {
	if = {
		limit = {
			OR = {
				has_idea = MTG_naval_treaty_adherent
				has_idea = MTG_naval_treaty_cheating
				has_idea = MTG_naval_treaty_adherent_reduced
			}
		}
		remove_ideas = MTG_naval_treaty_adherent
		remove_ideas = MTG_naval_treaty_cheating
		remove_ideas = MTG_naval_treaty_adherent_reduced
	}
}

# War economy at start if high WS
hom_set_war_economy_effect = {
	if = {
		limit = {
			has_war_support > 0.5
			OR = {
				has_idea = isolation
				has_idea = civilian_economy
				has_idea = low_economic_mobilisation
				has_idea = partial_economic_mobilisation
			}
		}
		add_ideas = war_economy
	}
}

# === Technology unlocks ===

# Air techs up to ~1940
hom_unlock_air_techs_1940_effect = {
	if = {
		limit = { has_dlc = "By Blood Alone" }
		# BBA airframes + key modules
		set_technology = {
			iw_small_airframe = 1
			basic_small_airframe = 1
			improved_small_airframe = 1
			iw_medium_airframe = 1
			basic_medium_airframe = 1
			improved_medium_airframe = 1
			iw_large_airframe = 1
			basic_large_airframe = 1
			engines_1 = 1
			engines_2 = 1
			air_torpedoe_1 = 1
			air_torpedoe_2 = 1
			popup = no
		}
	}
	else = {
		# Vanilla planes
		set_technology = {
			early_fighter = 1
			fighter1 = 1
			fighter2 = 1
			CAS1 = 1
			CAS2 = 1
			naval_bomber1 = 1
			naval_bomber2 = 1
			tactical_bomber1 = 1
			tactical_bomber2 = 1
			strategic_bomber1 = 1
			early_bomber = 1
			popup = no
		}
	}
	# Common radar
	complete_special_project = sp:sp_air_radar
	set_technology = {
		radio_detection = 1
		centimetric_radar = 1
		popup = no
	}
}
