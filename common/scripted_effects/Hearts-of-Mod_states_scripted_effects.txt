transfer_state_to_conditionnel = {
	effect_tooltip = {
		transfer_state_to = var:dest_owner
	}
	hidden_effect = {
		if = {
			limit = {
				is_fully_controlled_by = OWNER
			}
			transfer_state_to = var:dest_owner
		}
		else = {
			set_state_owner_to = var:dest_owner
		}
	}
}

			}
