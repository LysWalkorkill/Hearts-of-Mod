on_actions = {
	on_startup = {
		# Ajout de noyau Belarus à Vilna pour éviter un border gore si la Pologne est démantelé
		effect = {
			784 = {
				add_core_of = BLR
			}
		}
	}
	#ROOT: New controller; FROM: Old controller; FROM.FROM: State
	on_state_control_changed = {
		# Chute de Moscou
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = hearts_of_mod_fall_of_moscow
						option = ENABLED
					}
					FROM.FROM = {
						is_owned_by = FROM
						state = 219 # Moscow
					}
				}
				if = {
					limit = {
						FROM = {
							any_character = {
								OR = {
									is_character = SOV_Mita
									is_character = SOV_Katyusha
								}
								NOT = {
									has_character_flag = is_fired
									has_character_flag = is_arrested
									has_character_flag = is_killed
								}
							}
							original_tag = SOV
							is_subject = no
							has_country_leader_ideology = stalinism
							NOT = {
								has_country_flag = Hearts-of-Mod_SOV_Chute_Moscou_leader_on_action
							}
						}
						ROOT = {
							has_war_with = FROM
						}
					}
					FROM = {
						save_event_target_as = this_country
						random_character = {
							limit = {
								OR = {
									is_character = SOV_Mita
									is_character = SOV_Katyusha
								}
								NOT = {
									has_character_flag = is_fired
									has_character_flag = is_arrested
									has_character_flag = is_killed
								}
							}
							save_event_target_as = this_character
							unit_leader_event = Hearts-of-Mod_SOV_Chute_Moscou.1
						}
					}
				}
			}
		}
	}
	on_weekly = {
		# Si un pays qui possède aucun état mais contrôle un état, le faire disparaître
		effect = {
			every_state = {
				limit = {
					CONTROLLER = {
						check_variable = { THIS.num_owned_states < 1 }
					}
				}
				CONTROLLER = {
					delete_unit = {
						disband = yes
					}
				}
				set_state_controller_to = OWNER
			}
		}
		# Défini le controlleur vers le nouveau possédeur
		effect = {
			if = {
				limit = {
					any_owned_state = {
						NOT = {
							is_fully_controlled_by = ROOT
						}
						OR = {
							is_fully_controlled_by = CONTROLLER
							CONTROLLER = {
								has_war = no
							}
						}
						CONTROLLER = {
							NOT = {
								has_war_with = ROOT
							}
							OR = {
								tag = ROOT
								is_in_faction_with = ROOT
								is_subject_of = ROOT
								has_subject = ROOT
							}
						}
					}
				}
				every_owned_state = {
					limit = {
						NOT = {
							is_fully_controlled_by = ROOT
						}
						OR = {
							is_fully_controlled_by = CONTROLLER
							CONTROLLER = {
								has_war = no
							}
						}
						CONTROLLER = {
							NOT = {
								has_war_with = ROOT
							}
							OR = {
								tag = ROOT
								is_in_faction_with = ROOT
								is_subject_of = ROOT
								has_subject = ROOT
							}
						}
					}
					set_state_controller_to = ROOT
				}
			}
		}
	}
}
